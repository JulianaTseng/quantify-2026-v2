// --- 文件处理 ---
function handleFile(file) {
if (!file) return;
if (!file.name.toLowerCase().endsWith('.pdf')) return alert('Please upload a PDF file.');
state.file = file;
$('#fileName').innerText = file.name;
$('#fileStatus').style.display = 'block';
$('#resultSection').style.display = 'none';
$('#prog').innerHTML = '';
}
function wireDragDrop() {
const drop = $('#drop');
if (!drop) return;
['dragenter', 'dragover'].forEach(e => drop.addEventListener(e, ev => { ev.preventDefault(); drop.classList.add('over'); }, false));
['dragleave', 'drop'].forEach(e => drop.addEventListener(e, ev => { ev.preventDefault(); drop.classList.remove('over'); }, false));
drop.addEventListener('drop', e => { if (e.dataTransfer.files.length > 0) handleFile(e.dataTransfer.files[0]); }, false);
}
// --- 核心：本地提取 PDF -> 发送给自定义 API ---
async function runAnalysis() {
if (!state.file || state.isAnalyzing) return;
const cfg = loadCfg();
if (!cfg.apiKey || !cfg.endpoint || !cfg.model) {
return alert('Please configure Endpoint, API Key, and Model Name in Settings first.');
}
state.isAnalyzing = true;
const btn = $('#analyzeBtn');
btn.disabled = true; btn.innerText = 'Processing...';
const prog = $('#prog');
try {
// 1. 本地读取 PDF 文本
prog.innerHTML = `<div class="prog"><div class="msg">Step 1/2: Extracting text from PDF locally...</div></div>`;
const arrayBuffer = await state.file.arrayBuffer();
const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
let fullText = "";
// 为了速度，这里只提取前 10 页作为演示
const pagesToRead = Math.min(pdf.numPages, 10);
for (let i = 1; i <= pagesToRead; i++) {
const page = await pdf.getPage(i);
const textContent = await page.getTextContent();
fullText += textContent.items.map(item => item.str).join(" ") + "\n";
}
if (!fullText.trim()) {
throw new Error("No text found in PDF. It might be a scanned image.");
}
// 2. 发送给 API
prog.innerHTML = `<div class="prog"><div class="msg">Step 2/2: Sending to AI for analysis (${cfg.model})...</div></div>`;
// 处理 Endpoint 格式 (兼容带不带 /v1 或 /chat/completions)
let url = cfg.endpoint.replace(/\/$/, '');
if (!url.endsWith('/chat/completions')) {
url = url.endsWith('/v1') ? url + '/chat/completions' : url + '/v1/chat/completions';
}
const response = await fetch(url, {
method: 'POST',
headers: {
'Content-Type': 'application/json',
'Authorization': 'Bearer ' + cfg.apiKey
},
body: JSON.stringify({
model: cfg.model,
messages: [
{ role: "system", content: "You are a professional Quantity Surveyor assistant. Extract key information from the provided document text." },
{ role: "user", content: "Please summarize the core content, extract key figures, dates, and important clauses from this document:\n\n" + fullText }
],
temperature: 0.2
})
});
const data = await response.json();
if (!response.ok) {
throw new Error(data.error?.message || JSON.stringify(data));
}
if (data.choices && data.choices.length > 0) {
prog.innerHTML = `<div class="prog done">Analysis completed!</div>`;
$('#resultSection').style.display = 'block';
// 简单地将 Markdown 渲染为 HTML（依赖 AI 返回标准 markdown）
$('#aiResult').innerHTML = data.choices[0].message.content.replace(/\n/g, '<br>');
} else {
throw new Error('AI returned an empty response.');
}
} catch (error) {
console.error(error);
prog.innerHTML = `<div class="prog err">Error: ${error.message}</div>`;
} finally {
state.isAnalyzing = false; btn.disabled = false; btn.innerText = 'Analyze with AI';
}
}

async function testConnection() {
const endpoint = $('#cfg-endpoint').value.trim();
const apiKey = $('#cfg-key').value.trim();
const model = $('#cfg-model').value.trim();
const resDiv = $('#test-result');
if (!endpoint || !apiKey || !model) {
resDiv.style.display = 'block'; resDiv.style.color = 'var(--bad)'; resDiv.innerText = 'Please fill all fields first.';
return;
}
resDiv.style.display = 'block'; resDiv.style.color = 'var(--muted)'; resDiv.innerText = 'Testing...';
try {
let url = endpoint.replace(/\/$/, '');
if (!url.endsWith('/chat/completions')) {
url = url.endsWith('/v1') ? url + '/chat/completions' : url + '/v1/chat/completions';
}
const response = await fetch(url, {
method: 'POST',
headers: {
'Content-Type': 'application/json',
'Authorization': 'Bearer ' + apiKey
},
body: JSON.stringify({
model: model,
messages: [{ role: "user", content: "Hello" }],
max_tokens: 5
})
});
if (response.ok) {
resDiv.style.color = 'var(--good)'; resDiv.innerText = '✅ Connection successful!';
} else {
const err = await response.json();
resDiv.style.color = 'var(--bad)'; resDiv.innerText = '❌ Failed: ' + (err.error?.message || response.status);
}
} catch (e) {
resDiv.style.color = 'var(--bad)'; resDiv.innerText = '❌ Network Error: ' + e.message;
}
}

closeModal();
}
